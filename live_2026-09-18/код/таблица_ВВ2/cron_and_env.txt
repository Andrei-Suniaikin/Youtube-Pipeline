## /root/ops/ww2_sheet/ (рабочая папка на VPS)
# run.sh                       обёртка: читает .env, flock -w 900, пишет в ww2_sheet.log
# ww2_sheet.py, ww2_pool.py    код (md5 = Mac ~/ops/*.py на 18.09)
# .env                         WW2_YT_KEY=<ключ YouTube Data API>            — СЕКРЕТ, в пакет не идёт
# google_service_account.json  ключ сервисного аккаунта Google Sheets        — СЕКРЕТ, в пакет не идёт
# pool_state.json (~4.5 МБ)    состояние пула: donors{<канал>: {full, seen}} — рабочие данные, не код
# ww2_names.json               переименования каналов для переноса в «Очередь»
# venv/                        gspread 6.2.1, google-auth 2.58.0

## Переменные окружения, которые читает код таблицы
# WW2_SHEET_ID   id Google-таблицы (в коде есть дефолт — в пакете заменён маркером)
# WW2_SA         путь к ключу сервисного аккаунта
# WW2_YT_KEY     ключ YouTube Data API (или WW2_YT_KEY_FILE — файл, из которого ключ достаётся регуляркой "key AIza...")
# WW2_REG        путь к ~/ops/registry.db (нужен только для --rebuild)
# WW2_NICHES     путь к ~/niches (нужен только для --rebuild)
# WW2_KEYS_FILE  файл ключей, из него берётся "openai" для разбора роликов (по умолчанию /root/video_studio_api_keys.json)
# WW2_OPENAI_KEY альтернатива WW2_KEYS_FILE
# WW2_POOL_MODEL модель разбора, по умолчанию gpt-4.1
# WW2_POOL_STATE путь к pool_state.json
